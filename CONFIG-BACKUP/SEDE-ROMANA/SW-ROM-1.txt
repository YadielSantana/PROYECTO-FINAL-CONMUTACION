enable
configure terminal
hostname SW-ROM-1

! Creación de VLANs
vlan 99
 name ADMIN-RED
vlan 210
 name CALL_CENTER
vlan 220
 name VENTAS
vlan 230
 name RECURSOS_HUMANOS
exit

! Configuración de SVI para administración (VLAN 99)
interface vlan 99
 ip address 192.168.4.162 255.255.255.240
 no shutdown
exit
ip default-gateway 192.168.4.161

! Puerto Troncal hacia el Router ROMANA
interface e0/0
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 99,210,220,230
 no shutdown

! Puertos Troncales hacia SW-ROM-2 (EtherChannel)
interface range e0/1-3
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 99,210,220,230
 channel-group 1 mode active
 no shutdown

interface port-channel 1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 99,210,220,230
exit
write memory
