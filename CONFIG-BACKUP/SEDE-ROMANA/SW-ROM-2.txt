enable
configure terminal
hostname SW-ROM-2

! Creación de VLANs
vlan 99
 name ADMIN-RED
vlan 210
 name CALL_CENTER
vlan 220
 name VENTAS
vlan 230
 name RECURSOS_HUMANOS
exit

! Configuración de SVI para administración (VLAN 99)
interface vlan 99
 ip address 192.168.4.163 255.255.255.240
 no shutdown
exit
ip default-gateway 192.168.4.161

! Puertos Troncales hacia SW-ROM-1 (EtherChannel)
interface range e0/1-3
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 99,210,220,230
 channel-group 1 mode active
 no shutdown

interface port-channel 1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 99,210,220,230

! Configuración de Puertos de Acceso
interface e1/0
 description VPC16 Call Center
 switchport mode access
 switchport access vlan 210
 spanning-tree portfast
 no shutdown

interface e1/1
 description VPC27 Ventas y Atencion al C
 switchport mode access
 switchport access vlan 220
 spanning-tree portfast
 no shutdown

interface e1/2
 description VPC28 Recursos Humanos
 switchport mode access
 switchport access vlan 230
 spanning-tree portfast
 no shutdown
exit
DO write memory
