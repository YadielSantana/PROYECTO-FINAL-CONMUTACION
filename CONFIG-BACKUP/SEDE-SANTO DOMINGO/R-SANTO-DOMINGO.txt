! --- 2. Loopback para DHCP Relay / router-id estable ---
interface Loopback0
 description DHCP-Relay-Target
 ip address 192.168.9.1 255.255.255.255

! --- 3. Enlace ruteado hacia SW-SD-1 (multicapa) ---
interface Ethernet0/1
 description Routed Link to SW-SD-1
 no shutdown
 ip address 192.168.4.145 255.255.255.252
 ip nat inside
 ip tcp adjust-mss 1360

! --- 4. Enlace ruteado hacia SW-SD-2 (multicapa) ---
interface Ethernet0/2
 description Routed Link to SW-SD-2
 no shutdown
 ip address 192.168.4.149 255.255.255.252

 ip tcp adjust-mss 1360
 duplex full

! --- 5. Pools DHCP para los Clientes (relay desde SVIs de los switches) ---

ip dhcp excluded-address 192.168.2.129 192.168.2.131
ip dhcp excluded-address 192.168.2.1 192.168.2.3
ip dhcp excluded-address 192.168.1.1 192.168.1.3
