ip routing

! Crear todas las VLANs
vlan 10,20,30,40,99
exit

! Objeto de tracking sobre el enlace hacia el router
track 1 interface Ethernet0/0 line-protocol

! Port-Channel inter-switch (e0/1 - e0/3) hacia SW-SD-2
interface range Ethernet0/1 - 3
 description Channel-Group to SW-SD-2
 switchport trunk encapsulation dot1q
 switchport mode trunk
 channel-group 1 mode active
 duplex full
 no shutdown

interface Port-channel 1
 description Inter-Switch Link (L2)
 switchport trunk encapsulation dot1q
 switchport mode trunk
 no shutdown

! Enlace ruteado hacia R-SANTO-DOMINGO (antes trunk, ahora L3)
interface Ethernet0/0
 description Routed Link to R-SANTO-DOMINGO
 no switchport
 ip address 192.168.4.146 255.255.255.252
 duplex full
 no shutdown

! Puertos enlace descendente hacia switches de acceso
interface range Ethernet1/0 - 2
 description Downlinks to SW-SD-3, SW-SD-4, SW-SD-5
 switchport trunk encapsulation dot1q
 switchport mode trunk
 duplex full
 no shutdown

! --- SVIs + HSRP + DHCP Relay ---
interface Vlan10

 standby 10 track 1 decrement 60

 standby 30 track 1 decrement 60
 no shutdown

 standby 40 track 1 decrement 60
 no shutdown
