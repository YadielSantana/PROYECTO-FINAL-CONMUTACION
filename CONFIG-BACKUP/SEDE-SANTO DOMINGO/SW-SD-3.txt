enable
configure terminal
hostname SW-SD-3

! Crear VLANs
vlan 10
 name Call-Center
vlan 20
 name Ventas-Atencion
vlan 99
 name ADMIN-RED
exit

! Puertos de acceso a los VPCs
interface Ethernet2/0
 description VPC VLAN 10
 switchport mode access
 switchport access vlan 10
 no shutdown

interface Ethernet2/1
 description VPC VLAN 20
 switchport mode access
 switchport access vlan 20
 no shutdown

! Troncales hacia Core/Distribución
interface range Ethernet0/0 - 1
 description Trunk to SW-SD-1 & SW-SD-2
 switchport trunk encapsulation dot1q   ! (Si tus Cisco IOU/IOL lo requieren)
 switchport mode trunk
 switchport trunk allowed vlan 10,20,99
 no shutdown
exit
